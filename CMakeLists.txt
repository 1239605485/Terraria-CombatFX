cmake_minimum_required(VERSION 3.22)
project(CombatFX VERSION 0.4.3 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(ANDROID)
  enable_language(CXX)
  if(NOT EXISTS "${COMBATFX_DOBBY_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Set COMBATFX_DOBBY_SOURCE_DIR to the pinned TEFKernel libs/dobby directory")
  endif()
  set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
  set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
  set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
  set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
  add_subdirectory("${COMBATFX_DOBBY_SOURCE_DIR}" build-dobby EXCLUDE_FROM_ALL)
  add_library(combatfx SHARED src/combatfx.c src/vfx_image.c)
  target_include_directories(combatfx PRIVATE include src)
  target_compile_options(combatfx PRIVATE -Wall -Wextra -Werror -O2 -fno-common)
  target_link_options(combatfx PRIVATE -Wl,--no-undefined -Wl,-Bsymbolic -Wl,-z,max-page-size=16384)
  target_link_libraries(combatfx PRIVATE dobby_static log dl m)
else()
  enable_testing()
  add_executable(limits_test tests/limits_test.c)
  target_include_directories(limits_test PRIVATE src)
  target_compile_options(limits_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME effect_budget COMMAND limits_test)
endif()
if(NOT ANDROID)
  add_executable(runtime_test tests/runtime_test.c tests/hook_stubs.c src/vfx_image.c)
  target_compile_definitions(runtime_test PRIVATE __ANDROID__)
  target_include_directories(runtime_test PRIVATE include src)
  target_compile_options(runtime_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(runtime_test PRIVATE m)
  add_test(NAME runtime_guards COMMAND runtime_test)
endif()
if(NOT ANDROID)
  add_executable(combo_test tests/combo_test.c)
  target_include_directories(combo_test PRIVATE src)
  target_compile_options(combo_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(combo_test PRIVATE m)
  add_test(NAME combo_geometry COMMAND combo_test)
  add_executable(wood_runtime_test tests/wood_runtime_test.c tests/hook_stubs.c src/vfx_image.c)
  target_compile_definitions(wood_runtime_test PRIVATE __ANDROID__)
  target_include_directories(wood_runtime_test PRIVATE include src)
  target_compile_options(wood_runtime_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(wood_runtime_test PRIVATE m)
  add_test(NAME wood_runtime_guards COMMAND wood_runtime_test)
endif()

if(NOT ANDROID)
  add_executable(vfx_image_test tests/vfx_image_test.c src/vfx_image.c)
  target_include_directories(vfx_image_test PRIVATE include src)
  target_compile_options(vfx_image_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(vfx_image_test PRIVATE m)
  add_test(NAME actual_png_textures COMMAND vfx_image_test "${CMAKE_CURRENT_SOURCE_DIR}/assets/textures")
endif()
